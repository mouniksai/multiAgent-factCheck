% =============================================================================
\begin{frame}
    \framesubtitle{SYSTEM OVERVIEW}
    \frametitle{Multi-Agent RAG for Reliable Claim Verification}
    \vspace{-0.05cm}
    {\small
    \begin{itemize}\setlength\itemsep{0.08cm}
        \item \textbf{Empirical Grounding via Multi-Agent RAG:} Designed to verify complex real-world factual claims by executing live web retrieval across academic and institutional repositories.
        \item \textbf{Atomic Sub-Claim Normalization:} Decomposes compound, multi-clause assertions into 1--3 atomic propositions, eliminating multi-hop retrieval errors and query ambiguity.
        \item \textbf{Adversarial Dialectic Arena:} PRO and CON debater agents independently execute polarized search queries to construct grounded arguments and counter-arguments in iterative rounds.
        \item \textbf{Dialectical Cross-Examination:} An impartial Cross-Examiner audits debater turns for formal reasoning fallacies (\textit{Correlation vs.\ Causation, Hasty Generalization}) and isolates consensus vs.\ contention.
        \item \textbf{Deterministic Citation Guardrail:} Regex verifies that 100\% of cited evidence tags correspond to registered sources in the evidence pool prior to judicial arbitration.
        \item \textbf{Calibrated Epistemic Judgment:} An independent Supreme Judge model weighs verified evidence and domain authority to output calibrated verdicts with explicit uncertainty boundaries.
    \end{itemize}
    }
\end{frame}
